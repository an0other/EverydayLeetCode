class Solution {
    public int[] plusOne(int[] digits) {
        
        int mark = 1, n = digits.length - 1;
        while (mark != 0 && n >= 0) {
            int tmp = digits[n] + 1;

            digits[n] = tmp % 10;
            
            if (tmp >= 10) {
                n--;
            } else mark = 0;
        } 

        if (mark != 0){
            int[] ans = new int[digits.length+1];
            ans[0] = 1;
            for (int i = 1; i< ans.length; i++) ans[i] = digits[i-1];
            return ans;
        }
        return digits;
    }
}
